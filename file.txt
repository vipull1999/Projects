package COLLECTIONS;

import java.util.*;

class Banking {
    private String name;
    private int accntno;
    private int Balance;

    Banking(String name, int acctno, int Balance) {
        this.name = name;
        this.accntno = acctno;
        this.Balance = Balance;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public int getAccntno() {
        return accntno;
    }

    public void setAccntno(int accntno) {
        this.accntno = accntno;
    }

    public int getBalance() {
        return Balance;
    }

    public void setBalance(int balance) {
        Balance = balance;
    }
}

public class Crud {
    public static void main(String[] args) {
        ArrayList<Banking> list = new ArrayList<Banking>();
        Scanner sc = new Scanner(System.in);
        int ch;

        do {
            System.out.println("Enter your choice 1.Adding account 2. Display account 3.Update account 4.Delete account");
            ch = sc.nextInt();

            switch (ch) {
                case 1:
                    System.out.println("Enter customer details");
                    String name;
                    System.out.println("enter the name");
                    name = sc.next();
                    int accntno;
                    System.out.println("enter accnt no.");
                    accntno = sc.nextInt();
                    int Balance;
                    System.out.println("Enter the balance");
                    Balance = sc.nextInt();
                    list.add(new Banking(name, accntno, Balance));
                    System.out.println("account added succesfully");
                    break;

                case 2:
                    System.out.println("Display account");
                    for (Banking b : list) {
                        System.out.println(b.getName() + " " + b.getAccntno() + " " + b.getBalance());
                    }
                    break;
                case 3:
                    System.out.println("Enter old account number:");
                    int oldAccntno = sc.nextInt();

                    System.out.println("Enter new name:");
                    String new_name = sc.next();
                    System.out.println("account number will be same:");
                    
                    // int new_accntno = sc.nextInt();
                    System.out.println("Enter new balance:");
                    int new_Balance = sc.nextInt();

                    ListIterator<Banking> it = list.listIterator();
                    boolean flag = false;
                    while (it.hasNext()) {
                        Banking account = it.next();
                        if (account.getAccntno() == oldAccntno) {
                            it.set(new Banking(new_name, account.getAccntno(), new_Balance));
                            flag = true;
                            System.out.println("Details updated succesfully");
                            System.out.println("Account number: " + account.getAccntno());
                            break;
                        }
                    }
                    if (!flag) {
                        System.out.println("Account number not found.");
                    }
                    
                    break;
                case 4:
                    System.out.println("Enter account number to delete:");
                    int deleteAccntno = sc.nextInt();

                    ListIterator<Banking> deleteIterator = list.listIterator();
                    boolean deleted = false;
                    while (deleteIterator.hasNext()) {
                        Banking account = deleteIterator.next();
                        if (account.getAccntno() == deleteAccntno) {
                            deleteIterator.remove();
                            deleted = true;
                            break;
                        }
                    }
                    if (!deleted) {
                        System.out.println("Account number not found.");
                    }
                    break;
            }
        } while (ch != 0);
    }
}
